   328→/**

   334→ * Two ways to qualify. A window narrow enough that the player fills it is one:

   357→/**
   358→ * Whether [windowWidth] is enough to keep the player open beside the page rather
   359→ * than raising it over one: the least the page can live in and the least the
   360→ * player can live in, side by side. Stated as the sum of the two minimums rather
   361→ * than as a number of its own, so it cannot drift out of step with either.
   362→ *
   363→ * [windowWidth] is the width of the *window*, and it has to be measured rather
   364→ * than read off `Configuration.screenWidthDp` — in a freeform or desktop window
   365→ * that can report the display instead of the window, and it lands a beat late
   366→ * when the window is dragged. Deciding a two-pane split from a width the app
   367→ * does not have splits it at the wrong moment and in the wrong place.
   368→ *
   369→ * Public because it is not only the player's business: the page it stands next
   370→ * to loses the mini player, gets its bottom inset back, and stops being able to
   371→ * raise the sheet at all.
   372→ */
   373→fun dockedPlayerAvailable(windowWidth: Dp): Boolean =
   374→    windowWidth >= DOCKED_PAGE_MIN_WIDTH + DOCKED_PLAYER_MIN_WIDTH

   376→/** How wide that pane is. Only meaningful where [dockedPlayerAvailable] is true. */
   377→fun dockedPlayerWidth(windowWidth: Dp): Dp =
   378→    (windowWidth * DOCKED_PLAYER_FRACTION)
   379→        .coerceIn(DOCKED_PLAYER_MIN_WIDTH, DOCKED_PLAYER_MAX_WIDTH)
   380→        // Never at the page's expense. The floor above is what the player wants;
   381→        // this is what it may actually have, and where the two disagree the page
   382→        // wins — [dockedPlayerAvailable] is the promise that they only disagree
   383→        // in windows narrow enough that there is no pane at all.
   384→        .coerceAtMost(windowWidth - DOCKED_PAGE_MIN_WIDTH)
