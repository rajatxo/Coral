   581→                    else stringResource(R.string.format_multiplier, "%.2f".format(it))
   582→                },
   583→                steps = 39,
   584→                label = stringResource(R.string.playback_speed)
   585→            )
   586→            SliderDialogBody(
   587→                provideState = { remember(pitch) { mutableFloatStateOf(pitch) } },
   588→                onSlideComplete = { pitch = it },
   589→                min = 0f,
   590→                max = 2f,
   591→                toDisplay = {
   592→                    if (it <= 0.01f) stringResource(R.string.minimum_speed_value)
   593→                    else stringResource(R.string.format_multiplier, "%.2f".format(it))
   594→                },
   595→                steps = 39,
   596→                label = stringResource(R.string.playback_pitch)
   597→            )
   598→            Box(
   599→                modifier = Modifier.fillMaxWidth(),                contentAlignment = Alignment.Center
   600→            ) {
   601→                SecondaryTextButton(
   602→                    text = stringResource(R.string.reset),
   603→                    onClick = {
   604→                        speed = 1f
   605→                        pitch = 1f
   606→                    }
   607→                )
   608→            }
   609→        }
   610→
   611→        if (boostDialogOpen) {
   612→            fun submit(state: Float) = transaction {
   613→                mediaItem?.mediaId?.let { mediaId ->
   614→                    Database.setLoudnessBoost(
   615→                        songId = mediaId,
   616→                        loudnessBoost = state.takeUnless { it == 0f }
   617→                    )
   618→                }
   619→            }
   620→
   621→            SliderDialog(
   622→                onDismiss = { boostDialogOpen = false },
   623→                title = stringResource(R.string.volume_boost)
   624→            ) {
   625→                SliderDialogBody(
   626→                    provideState = {
   627→                        val state = remember { mutableFloatStateOf(0f) }
   628→                        LaunchedEffect(mediaItem) {
   629→                            mediaItem?.mediaId?.let { mediaId ->
   630→                                Database
   631→                                    .loudnessBoost(mediaId)
   632→                                    .distinctUntilChanged()
   633→                                    .collect { state.floatValue = it ?: 0f }
   634→                            }
   635→                        }
   636→                        state
   637→                    },
   638→                    onSlideComplete = { submit(it) },
   639→                    min = -20f,
   640→                    max = 20f,
   641→                    toDisplay = { stringResource(R.string.format_db, "%.2f".format(it)) }
   642→                )
   643→                Box(
   644→                    modifier = Modifier.fillMaxWidth(),
   645→                    contentAlignment = Alignment.Center
   646→                ) {
   647→                    SecondaryTextButton(
   648→                        text = stringResource(R.string.reset),
   649→                        onClick = { submit(0f) }
   650→                    )
   651→                }
   652→            }
   653→        }
   654→
   655→        if (binder != null) Queue(
   656→            layoutState = playerBottomSheetState,
   657→            binder = binder,
   658→            beforeContent = {
   659→                if (playerLayout == PlayerPreferences.PlayerLayout.New) IconButton(
   660→                    onClick = { trackLoopEnabled = !trackLoopEnabled },
   661→                    icon = R.drawable.infinite,
   662→                    enabled = trackLoopEnabled,
   663→                    modifier = Modifier
   664→                        .padding(vertical = 8.dp)
   665→                        .size(20.dp)
   666→                ) else Spacer(modifier = Modifier.width(20.dp))
   667→            },
   668→            afterContent = {
   669→                IconButton(
   670→                    icon = R.drawable.ellipsis_horizontal,
   671→                    color = colorPalette.text,
   672→                    onClick = {
   673→                        mediaItem?.let {
   674→                            menuState.display {
   675→                                PlayerMenu(
   676→                                    onDismiss = menuState::hide,
   677→                                    mediaItem = it,
   678→                                    binder = binder,
   679→                                    onShowSpeedDialog = { audioDialogOpen = true },
   680→                                    onShowNormalizationDialog = {
   681→                                        boostDialogOpen = true
   682→                                    }.takeIf { volumeNormalization }
   683→                                )
   684→                            }
   685→                        }
   686→                    },
   687→                    modifier = Modifier
   688→                        .padding(vertical = 8.dp)
   689→                        .size(20.dp)
   690→                )
   691→            },
   692→            modifier = Modifier.align(Alignment.BottomCenter),
   693→            shape = shape
   694→        )
   695→      }
   696→    }
   697→}

   699→@Composable
   700→@OptIn(UnstableApi::class)
   701→private fun PlayerMenu(
   702→    binder: PlayerService.Binder,
   703→    mediaItem: MediaItem,
   704→    onDismiss: () -> Unit,
   705→    onShowSpeedDialog: (() -> Unit)? = null,
   706→    onShowNormalizationDialog: (() -> Unit)? = null
   707→) {
   708→    val launchEqualizer by rememberEqualizerLauncher(audioSessionId = { binder.player.audioSessionId })
   709→
   710→    BaseMediaItemMenu(
   711→        mediaItem = mediaItem,
   712→        onStartRadio = {
   713→            binder.stopRadio()
   714→            binder.player.seamlessPlay(mediaItem)
   715→            binder.setupRadio(NavigationEndpoint.Endpoint.Watch(videoId = mediaItem.mediaId))
   716→        },
   717→        onGoToEqualizer = launchEqualizer,
   718→        onShowSleepTimer = {},
   719→        onDismiss = onDismiss,
   720→        onShowSpeedDialog = onShowSpeedDialog,
   721→        onShowNormalizationDialog = onShowNormalizationDialog
   722→    )
   723→}
   724→
   725→private fun onDismiss(binder: PlayerService.Binder) {
   726→    binder.stopRadio()
   727→    binder.player.clearMediaItems()
   728→}
