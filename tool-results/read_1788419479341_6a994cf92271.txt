    92→private const val INLINE_LYRIC_UPDATE_DELAY = 50L
    93→private const val LYRIC_SEARCH_PHRASE_INTERVAL_MS = 1800L
    94→
    95→private val LYRIC_SEARCH_PHRASES = listOf(
    96→    "Almost words",
    97→    "Still looking",
    98→    "Matching the track",
    99→    "One more source to check"
   100→)
   101→
   102→@Composable
   103→fun BitChordPlayer(
   104→    mediaItem: MediaItem,
   105→    binder: PlayerService.Binder,
   106→    shouldBePlaying: Boolean,
   107→    position: Long,
   108→    duration: Long,
   109→    likedAt: Long?,
   110→    setLikedAt: (Long?) -> Unit,
   111→    isShowingLyrics: Boolean,
   112→    onShowLyrics: (Boolean) -> Unit,
   113→    onOpenQueue: () -> Unit,
   114→    onTitleClick: (() -> Unit)? = null,
   115→    onArtistClick: (() -> Unit)? = null,
   116→    onOpenLyricsDialog: () -> Unit = {},
   117→    modifier: Modifier = Modifier
   118→) {
   119→    val (colorPalette, typography) = LocalAppearance.current
   120→    val metadata = mediaItem.mediaMetadata
   121→    val media = remember(mediaItem, duration) { mediaItem.toUiMedia(duration) }
   122→
   123→    // Subtle scale pulse on the background album art when play/pause toggles
   124→    val artScale by animateFloatAsState(
   125→        targetValue = if (shouldBePlaying) 1f else 0.96f,
   126→        animationSpec = spring(dampingRatio = Spring.DampingRatioLowBouncy),
   127→        label = "artScale"
   128→    )
   129→
   130→    var shuffleOn by remember { mutableStateOf(binder.player.shuffleModeEnabled) }
   131→    var repeatMode by remember { mutableStateOf(binder.player.repeatMode) }
   132→
   133→    // --- inline synced lyric line, with background auto-fetch ---
   134→    var storedLyrics by remember(mediaItem.mediaId) { mutableStateOf<LyricsData?>(null) }
   135→    var isFetchingLyrics by remember(mediaItem.mediaId) { mutableStateOf(false) }
   136→
   137→    LaunchedEffect(mediaItem.mediaId) {
   138→        runCatching {
   139→            withContext(Dispatchers.IO) {
   140→                Database
   141→                    .lyrics(mediaItem.mediaId)
   142→                    .distinctUntilChanged()
   143→                    .cancellable()
   144→                    .collect { currentLyrics ->
   145→                        storedLyrics = currentLyrics
   146→
   147→                        if (currentLyrics?.fixed != null && currentLyrics.synced != null) {
   148→                            isFetchingLyrics = false
   149→                            return@collect
   150→                        }
   151→
   152→                        isFetchingLyrics = true
   153→
   154→                        val album = metadata.albumTitle?.toString()
   155→                        val artist = metadata.artist?.toString().orEmpty()
   156→                        val title = metadata.title?.toString().orEmpty().let {
   157→                            if (mediaItem.mediaId.startsWith(LOCAL_KEY_PREFIX)) it
   158→                                .substringBeforeLast('.')
   159→                                .trim()
   160→                            else it
   161→                        }
   162→                        val strippedTitle = title.split("(")[0].trim()
   163→
   164→                        coroutineScope {
   165→                            val durationDeferred = async {
   166→                                var d = withContext(Dispatchers.Main) {
   167→                                    binder.player.duration.takeIf { it > 0 } ?: C.TIME_UNSET
   168→                                }
   169→                                while (d == C.TIME_UNSET) {
   170→                                    delay(100)
   171→                                    d = withContext(Dispatchers.Main) {
   172→                                        binder.player.duration.takeIf { it > 0 } ?: C.TIME_UNSET
   173→                                    }
   174→                                }
   175→                                d
   176→                            }
   177→
   178→                            val innertube = async(Dispatchers.IO) {
   179→                                runCatching {
   180→                                    Innertube.lyrics(NextBody(videoId = mediaItem.mediaId))?.getOrNull()
   181→                                }.getOrNull()
   182→                            }
   183→
   184→                            val fixed = currentLyrics?.fixed ?: innertube.await() ?: run {
   185→                                val d = durationDeferred.await()
   186→                                runCatching {
   187→                                    LrcLib.bestLyrics(
   188→                                        artist = artist,
   189→                                        title = title,
   190→                                        duration = d.milliseconds,
   191→                                        album = album,
   192→                                        synced = false
   193→                                    )?.map { it?.text }?.getOrNull()
   194→                                }.getOrNull()
   195→                            }
   196→
   197→                            val synced = currentLyrics?.synced ?: run {
   198→                                val d = durationDeferred.await()
   199→
   200→                                val lrcMain = async(Dispatchers.IO) {
   201→                                    runCatching {
   202→                                        LrcLib.bestLyrics(
   203→                                            artist = artist,
   204→                                            title = title,
   205→                                            duration = d.milliseconds,
   206→                                            album = album
   207→                                        )?.map { it?.text }?.getOrNull()
   208→                                    }.getOrNull()
   209→                                }
   210→                                val lrcRetry = async(Dispatchers.IO) {
   211→                                    runCatching {
   212→                                        LrcLib.bestLyrics(
   213→                                            artist = artist,
   214→                                            title = strippedTitle,
   215→                                            duration = d.milliseconds,
   216→                                            album = album
   217→                                        )?.map { it?.text }?.getOrNull()
   218→                                    }.getOrNull()
   219→                                }
   220→                                val kugou = async(Dispatchers.IO) {
   221→                                    runCatching {
   222→                                        KuGou.lyrics(
   223→                                            artist = artist,
   224→                                            title = title,
   225→                                            duration = d / 1000
   226→                                        )?.map { it?.value }?.getOrNull()
   227→                                    }.getOrNull()
   228→                                }
   229→                                lrcMain.await() ?: lrcRetry.await() ?: kugou.await()
   230→                            }
   231→
   232→                            LyricsData(
   233→                                songId = mediaItem.mediaId,
   234→                                fixed = fixed.orEmpty(),
   235→                                synced = synced.orEmpty()
   236→                            ).also {
   237→                                ensureActive()
   238→                                transaction {
   239→                                    runCatching {
   240→                                        Database.insert(mediaItem)
   241→                                        Database.upsert(it)
   242→                                    }
   243→                                }
   244→                            }
   245→                        }
   246→
   247→                        isFetchingLyrics = false
   248→                    }
   249→            }
   250→        }.exceptionOrNull()?.let {
   251→            if (it is CancellationException) throw it
   252→            isFetchingLyrics = false
   253→        }
   254→    }
   255→
   256→    val lyricsState = remember(storedLyrics) {
   257→        val file = storedLyrics?.synced?.takeIf { it.isNotBlank() }?.let {
   258→            LrcParser.parse(it)?.toLrcFile()
   259→        }
   260→
   261→        SynchronizedLyricsState(
   262→            sentences = file?.lines,
   263→            offset = file?.offset?.inWholeMilliseconds ?: 0L
   264→        )
   265→    }
   266→
   267→    val synchronizedLyrics = remember(lyricsState) {
   268→        lyricsState.sentences?.let {
   269→            SynchronizedLyrics(it.toImmutableMap()) {
   270→                binder.player.currentPosition + INLINE_LYRIC_UPDATE_DELAY + lyricsState.offset -
   271→                    (storedLyrics?.startTime ?: 0L)
   272→            }
   273→        }
   274→    }
   275→
   276→    LaunchedEffect(synchronizedLyrics) {
   277→        val current = synchronizedLyrics ?: return@LaunchedEffect
   278→        while (true) {
   279→            delay(INLINE_LYRIC_UPDATE_DELAY)
   280→            current.update()
   281→        }
   282→    }
   283→
   284→    val currentSentenceRaw = synchronizedLyrics?.let {
   285→        it.sentences.values.toImmutableList().getOrNull(it.index)
   286→    }
   287→    val currentLyricLine = currentSentenceRaw?.takeIf { it.isNotBlank() }
   288→    val isInstrumentalGap = currentSentenceRaw != null && currentSentenceRaw.isBlank()
   289→
   290→    var searchPhraseIndex by remember(mediaItem.mediaId) { mutableIntStateOf(0) }
   291→    LaunchedEffect(mediaItem.mediaId, isFetchingLyrics) {
   292→        if (!isFetchingLyrics) return@LaunchedEffect
   293→        while (true) {
   294→            delay(LYRIC_SEARCH_PHRASE_INTERVAL_MS)
   295→            searchPhraseIndex = (searchPhraseIndex + 1) % LYRIC_SEARCH_PHRASES.size
   296→        }
   297→    }
   298→
   299→    val showSearchingGlyph = isFetchingLyrics && currentLyricLine == null && !isInstrumentalGap
   300→    val showNoteGlyph = showSearchingGlyph || isInstrumentalGap
   301→    val lyricStripText = when {
   302→        currentLyricLine != null -> currentLyricLine
   303→        isInstrumentalGap -> "Instrumental"
   304→        isFetchingLyrics -> LYRIC_SEARCH_PHRASES[searchPhraseIndex]
   305→        else -> "Tap for lyrics"
   306→    }
   307→    // --- end inline synced lyric line ---
   308→
   309→    // ThinSlider state — local scrub value while user drags, null = follow player
   310→    var scrubValue by remember { mutableFloatStateOf(0f) }
   311→    var scrubbing by remember { mutableStateOf(false) }
   312→    val sliderValue = if (scrubbing) scrubValue
   313→                      else if (duration > 0) (position.toFloat() / duration).coerceIn(0f, 1f)
   314→                      else 0f
   315→
   316→    // ====================================================================
   317→    //  APPLE-MUSIC-STYLE NOW PLAYING
   318→    // ====================================================================
   319→    Box(modifier = modifier.fillMaxSize()) {
   320→        // 1. BLURRED BACKGROUND — full screen, blurred album art
   321→        AsyncImage(
   322→            model = metadata.artworkUri?.thumbnail(Dimensions.thumbnails.player.song.px),
   323→            contentDescription = null,
   324→            contentScale = ContentScale.Crop,
   325→            modifier = Modifier
   326→                .fillMaxSize()
   327→                .blur(32.dp)
   328→                .graphicsLayer {
   329→                    scaleX = artScale
   330→                    scaleY = artScale
   331→                }
   332→        )
   333→
   334→        // 2. CRISP TOP IMAGE — top 60% of screen, fades out smoothly into the
   335→        //    blurred background below. Fade is spread over a wide range (50%->80%
   336→        //    of image height) so there is NO hard seam.
   337→        Box(
   338→            modifier = Modifier
   339→                .fillMaxWidth()
   340→                .fillMaxHeight(0.60f)
   341→                .align(Alignment.TopCenter)
   342→        ) {
   343→            AsyncImage(
   344→                model = metadata.artworkUri?.thumbnail(Dimensions.thumbnails.player.song.px),
   345→                contentDescription = null,
   346→                contentScale = ContentScale.Crop,
   347→                modifier = Modifier
   348→                    .fillMaxSize()
   349→                    .drawWithContent {
   350→                        drawContent()
   351→                        // Long, gentle alpha mask: keep full opacity until 50% of
   352→                        // image height, then fade gradually to transparent at 100%.
   353→                        // This 50% wide fade range is what makes the blend seamless.
   354→                        drawRect(
   355→                            brush = Brush.verticalGradient(
   356→                                colorStops = arrayOf(
   357→                                    0.00f to Color.Black,
   358→                                    0.50f to Color.Black,
   359→                                    1.00f to Color.Transparent
   360→                                )
   361→                            ),
   362→                            blendMode = BlendMode.DstIn
   363→                        )
   364→                    }
   365→            )
   366→        }
   367→
   368→        // 3. DARK SCRIM (multi-stop, very gentle) — for text legibility on the
   369→        //    blurred bg. Starts nearly transparent at 30%, reaches full 75% black
   370→        //    at the very bottom. Distributes darkening across the whole height
   371→        //    so no hard transition line is visible.
   372→        Box(
   373→            modifier = Modifier
   374→                .fillMaxSize()
   375→                .background(
   376→                    Brush.verticalGradient(
   377→                        colorStops = arrayOf(
   378→                            0.00f to Color.Black.copy(alpha = 0.10f),
   379→                            0.30f to Color.Black.copy(alpha = 0.15f),
   380→                            0.50f to Color.Black.copy(alpha = 0.35f),
   381→                            0.70f to Color.Black.copy(alpha = 0.55f),
   382→                            1.00f to Color.Black.copy(alpha = 0.80f)
   383→                        )
   384→                    )
   385→                )
   386→        )
   387→
   388→        // 4. Full lyrics overlay (when user taps the lyric strip)
   389→        //    onOpenDialog is now wired to onOpenLyricsDialog so the expand icon
   390→        //    (top-left of lyrics overlay) opens the LrcLib search dialog.
   391→        Lyrics(
   392→            mediaId = mediaItem.mediaId,
   393→            isDisplayed = isShowingLyrics,
   394→            onDismiss = { onShowLyrics(false) },
   395→            ensureSongInserted = { Database.insert(mediaItem) },
   396→            mediaMetadataProvider = { mediaItem.mediaMetadata },
   397→            durationProvider = { binder.player.duration.takeIf { it > 0 } ?: C.TIME_UNSET },
   398→            onOpenDialog = onOpenLyricsDialog,
   399→            modifier = Modifier.fillMaxSize(),
   400→            shouldShowSynchronizedLyrics = PlayerPreferences.isShowingSynchronizedLyrics,
   401→            setShouldShowSynchronizedLyrics = { PlayerPreferences.isShowingSynchronizedLyrics = it },
   402→            showControls = true
   403→        )
   404→
   405→        // 5. Top drag-handle pill
   406→        Box(
   407→            modifier = Modifier
   408→                .align(Alignment.TopCenter)
   409→                .padding(top = 12.dp)
   410→                .width(40.dp)
   411→                .height(4.dp)
   412→                .clip(RoundedCornerShape(2.dp))
   413→                .background(Color.White.copy(alpha = 0.4f))
   414→        )
   415→
   416→        // 6. Main content column — bottom aligned, sits on top of blurred bg + scrim
   417→        Column(
   418→            horizontalAlignment = Alignment.CenterHorizontally,
   419→            verticalArrangement = Arrangement.Bottom,
   420→            modifier = Modifier
   421→                .fillMaxSize()
   422→                .padding(horizontal = 24.dp)
   423→                .padding(top = 40.dp, bottom = 32.dp)
   424→        ) {
   425→            // ---- Title + Artist + Heart row ----
   426→            Row(
   427→                horizontalArrangement = Arrangement.SpaceBetween,
   428→                verticalAlignment = Alignment.CenterVertically,
   429→                modifier = Modifier.fillMaxWidth()
   430→            ) {
   431→                Column(modifier = Modifier.weight(1f)) {
   432→                    BasicText(
   433→                        text = metadata.title?.toString().orEmpty(),
   434→                        style = typography.l.semiBold.copy(color = Color.White),
   435→                        maxLines = 1,
   436→                        overflow = TextOverflow.Ellipsis,
   437→                        modifier = Modifier.let {
   438→                            if (onTitleClick != null) it.clickable(onClick = onTitleClick) else it
   439→                        }
   440→                    )
   441→                    BasicText(
   442→                        text = metadata.artist?.toString().orEmpty(),
   443→                        style = typography.s.semiBold.copy(color = Color.White.copy(alpha = 0.7f)),
   444→                        maxLines = 1,
   445→                        overflow = TextOverflow.Ellipsis,
   446→                        modifier = Modifier.let {
   447→                            if (onArtistClick != null) it.clickable(onClick = onArtistClick) else it
   448→                        }
   449→                    )
   450→                }
   451→
   452→                // Heart inside a 40dp translucent circle
   453→                Box(
   454→                    modifier = Modifier
   455→                        .size(40.dp)
   456→                        .clip(RoundedCornerShape(50))
   457→                        .background(Color.White.copy(alpha = 0.2f))
   458→                        .clickable {
   459→                            setLikedAt(if (likedAt == null) System.currentTimeMillis() else null)
   460→                        },
   461→                    contentAlignment = Alignment.Center
   462→                ) {
   463→                    Icon(
   464→                        imageVector = if (likedAt == null) BitChordIcons.Heart
   465→                                      else BitChordIcons.HeartFilled,
   466→                        contentDescription = "Like",
   467→                        tint = if (likedAt != null) colorPalette.favoritesIcon else Color.White,
   468→                        modifier = Modifier.size(22.dp)
   469→                    )
   470→                }
   471→            }
   472→
   473→            Spacer(modifier = Modifier.height(16.dp))
   474→
   475→            // ---- Inline lyric strip ----
   476→            Row(
   477→                verticalAlignment = Alignment.CenterVertically,
   478→                modifier = Modifier
   479→                    .fillMaxWidth()
   480→                    .clickable { onShowLyrics(true) }
   481→                    .padding(vertical = 10.dp)
   482→            ) {
   483→                if (showNoteGlyph) {
   484→                    Image(
   485→                        imageVector = BitChordIcons.MusicNote,
   486→                        contentDescription = null,
   487→                        colorFilter = ColorFilter.tint(Color.White.copy(alpha = 0.6f)),
   488→                        modifier = Modifier.size(14.dp)
   489→                    )
   490→                    Spacer(modifier = Modifier.width(6.dp))
   491→                }
   492→                AnimatedContent(
   493→                    targetState = lyricStripText,
   494→                    transitionSpec = { fadeIn() togetherWith fadeOut() },
   495→                    label = "inlineLyricLine",
   496→                    modifier = Modifier.weight(1f, fill = false)
   497→                ) { line ->
   498→                    BasicText(
   499→                        text = line,
   500→                        style = typography.xs.semiBold.copy(color = Color.White.copy(alpha = 0.7f)),
   501→                        maxLines = 1,
   502→                        overflow = TextOverflow.Ellipsis
   503→                    )
   504→                }
   505→                Spacer(modifier = Modifier.width(4.dp))
   506→                Image(
   507→                    imageVector = BitChordIcons.ChevronRight,
   508→                    contentDescription = null,
   509→                    colorFilter = ColorFilter.tint(Color.White.copy(alpha = 0.7f)),
   510→                    modifier = Modifier.size(14.dp)
   511→                )
   512→            }
   513→
   514→            Spacer(modifier = Modifier.height(12.dp))
   515→
   516→            // ---- ThinSlider (Apple Music style) + timestamps ----
   517→            Row(
   518→                verticalAlignment = Alignment.CenterVertically,
   519→                horizontalArrangement = Arrangement.SpaceBetween,
   520→                modifier = Modifier.fillMaxWidth()
   521→            ) {
   522→                BasicText(
   523→                    text = formatTime(if (scrubbing) (scrubValue * duration).toLong() else position),
   524→                    style = typography.xs.semiBold.copy(color = Color.White.copy(alpha = 0.6f))
   525→                )
   526→                Spacer(modifier = Modifier.width(8.dp))
   527→                ThinSlider(
   528→                    value = sliderValue,
   529→                    onValueChange = {
   530→                        scrubbing = true
   531→                        scrubValue = it
   532→                    },
   533→                    onValueChangeFinished = {
   534→                        val targetMs = (scrubValue * duration).toLong()
   535→                        binder.player.seekTo(targetMs)
   536→                        scrubbing = false
   537→                    },
   538→                    modifier = Modifier.weight(1f)
   539→                )
   540→                Spacer(modifier = Modifier.width(8.dp))
   541→                BasicText(
   542→                    text = "-${formatTime((duration - (if (scrubbing) (scrubValue * duration).toLong() else position)).coerceAtLeast(0L))}",
   543→                    style = typography.xs.semiBold.copy(color = Color.White.copy(alpha = 0.6f))
   544→                )
   545→            }
   546→
   547→            Spacer(modifier = Modifier.height(20.dp))
   548→

   712→/** Format milliseconds as "M:SS" (e.g. 1:09). */
   713→private fun formatTime(ms: Long): String {
   714→    if (ms <= 0) return "0:00"
   715→    val totalSeconds = ms / 1000
   716→    val minutes = totalSeconds / 60
   717→    val seconds = totalSeconds % 60
   718→    return "%d:%02d".format(minutes, seconds)
   719→}
